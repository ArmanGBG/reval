    34→// ===== Helper: Convert number to Persian digits =====
    35→function toPersianDigits(num: number | string): string {
    36→  const persianDigits = ['۰', '۱', '۲', '۳', '۴', '۵', '۶', '۷', '۸', '۹'];
    37→  return num
    38→    .toString()
    39→    .split('')
    40→    .map((d) => persianDigits[parseInt(d)] ?? d)
    41→    .join('');
    42→}

   127→function DailyProgress({ completedCount, totalCount }: { completedCount: number; totalCount: number }) {
   128→  const percentage = totalCount > 0 ? Math.round((completedCount / totalCount) * 100) : 0;
   129→  const remaining = Math.max(0, totalCount - completedCount);
   130→
   131→  return (
   132→    <div className="surface-1 edge-highlight card-hover rounded-[var(--radius-lg)] p-4 md:p-5">
   133→      <div className="flex items-center justify-between mb-3">
   134→        <div className="flex items-center gap-2">

   136→            <Target className="w-4 h-4 text-[var(--accent)]" />
   137→          </div>
   138→          <div className="flex flex-col">
   139→            <span className="text-sm font-semibold text-[var(--foreground)]">پیشرفت امروز</span>
   140→            <span className="text-[11px] text-[var(--foreground-muted)]">
   141→              {toPersianDigits(completedCount)} از {toPersianDigits(totalCount)} وظیفه
   142→            </span>
   143→          </div>
   144→        </div>
   145→        <motion.span
   146→          key={percentage}
   147→          initial={{ opacity: 0, scale: 0.85 }}
   148→          animate={{ opacity: 1, scale: 1 }}
   149→          className="text-2xl md:text-3xl font-bold text-[var(--accent)] tabular-nums"
   150→        >
   151→          {toPersianDigits(percentage)}<span className="text-base">٪</span>
   152→        </motion.span>
   153→      </div>
   154→
   155→      {/* Progress bar */}
   156→      <div className="w-full h-2.5 bg-[var(--bg-overlay)] rounded-full overflow-hidden">
   157→        <motion.div
   158→          className="h-full rounded-full relative"
   159→          style={{
   160→            background: `linear-gradient(90deg, var(--accent) 0%, var(--accent-hover) 100%)`,
   161→          }}
   162→          initial={{ width: 0 }}
   163→          animate={{ width: `${percentage}%` }}
   164→          transition={{ duration: 0.8, ease: 'easeOut' }}
   165→        />
   166→      </div>
   167→
   168→      {remaining > 0 && (

   172→        </div>
   173→      )}
   174→      {remaining === 0 && totalCount > 0 && (
   175→        <div className="flex items-center gap-1.5 mt-3 text-[11px] text-[var(--accent)]">
   176→          <Check className="w-3.5 h-3.5" />
   177→          <span>هدف امروز کامل شد — برکانا! 🔥</span>
   178→        </div>
   179→      )}
   180→    </div>
   181→  );
   182→}

   463→function QuickStats({ tasks }: { tasks: Task[] }) {

   467→  const totalTests = todayTasks.reduce((sum, t) => sum + (t.actualTestCount ?? t.targetTestCount), 0);
   468→  const totalMinutes = todayTasks.reduce((sum, t) => sum + (t.actualTimeMinutes ?? t.targetTimeMinutes), 0);
   469→  const hours = Math.floor(totalMinutes / 60);
   470→  const mins = totalMinutes % 60;
   471→

   206→function FocusMusicTool() {

   208→
   209→  const handlePlay = useCallback(
   210→    (track: typeof tracks[0]) => {
   211→      if (currentTrack?.id === track.id && isPlaying) {
   212→        setIsPlaying(false);
   213→      } else {
   214→        setCurrentTrack(track);
   215→        setIsPlaying(true);
   216→      }
   217→    },
   218→    [currentTrack, isPlaying, setCurrentTrack, setIsPlaying]
   219→  );
   220→
   221→  const handleNext = useCallback(() => {
   222→    const idx = tracks.findIndex((t) => t.id === currentTrack?.id);
   223→    const next = tracks[(idx + 1) % tracks.length];
   224→    setCurrentTrack(next);
   225→    setIsPlaying(true);
   226→  }, [tracks, currentTrack, setCurrentTrack, setIsPlaying]);
   227→
   228→  return (
   229→    <div>
   230→      {/* Now Playing Hero */}
   231→      {currentTrack && (

   233→          initial={{ opacity: 0, scale: 0.95 }}

   240→            {currentTrack.cover}

   377→type MasteryFilter = 'all' | 'mastered' | 'review' | 'weak';
   378→
   379→function FlashcardsTool() {
   380→  const { flashcards, addFlashcard, updateFlashcard } = useAppStore();
   381→  const [tab, setTab] = useState<'study' | 'marked'>('study');
   382→  const [selectedSubject, setSelectedSubject] = useState<string | null>(null);
   383→  const [masteryFilter, setMasteryFilter] = useState<MasteryFilter>('all');
   384→  const [currentIndex, setCurrentIndex] = useState(0);
   385→  const [isFlipped, setIsFlipped] = useState(false);
   386→  const [showAddForm, setShowAddForm] = useState(false);
   387→  const [newFront, setNewFront] = useState('');
   388→  const [newBack, setNewBack] = useState('');
   389→  const [newSubject, setNewSubject] = useState('');
   390→
   391→  // Get unique subjects from all flashcards
   392→  const allSubjects = [...new Set(flashcards.map((c) => c.subject).filter(Boolean) as string[])];
   393→
   394→  // Filter cards based on all criteria
   395→  const filteredCards = flashcards.filter((c) => {
   396→    if (tab === 'marked') {
   397→      // Show only marked cards in the "marked" tab
   398→      if (c.mastery === 'mastered' && masteryFilter === 'all') return true;
   399→      if (c.mastery === 'review' && masteryFilter === 'all') return true;
   400→      if (c.mastery === 'weak' && masteryFilter === 'all') return true;
   401→      if (masteryFilter !== 'all' && c.mastery !== masteryFilter) return false;
   402→    }
   403→    // Subject filter
   404→    if (selectedSubject && c.subject !== selectedSubject) return false;
   405→    // Mastery filter
   406→    if (masteryFilter !== 'all' && c.mastery !== masteryFilter) return false;
   407→    return true;
   408→  });
   409→
   410→  const currentCard = filteredCards[currentIndex] || null;
   411→
   412→  const handleFlip = useCallback(() => {
   413→    setIsFlipped((prev) => !prev);
   414→  }, []);
   415→
   416→  const handleMastery = useCallback(
   417→    (mastery: Flashcard['mastery']) => {
   418→      if (!currentCard) return;
   419→      updateFlashcard(currentCard.id, { mastery });
   420→      setIsFlipped(false);
   421→      setTimeout(() => {
   422→        if (currentIndex < filteredCards.length - 1) {
   423→          setCurrentIndex((prev) => prev + 1);
   424→        }
   425→      }, 300);
   426→    },
   427→    [currentCard, updateFlashcard, currentIndex, filteredCards.length]
   428→  );
   429→
   430→  const handlePrev = useCallback(() => {
   431→    setIsFlipped(false);
   432→    setCurrentIndex((prev) => Math.max(0, prev - 1));
   433→  }, []);
   434→
   435→  const handleNext = useCallback(() => {
   436→    setIsFlipped(false);
   437→    setCurrentIndex((prev) => Math.min(filteredCards.length - 1, prev + 1));
   438→  }, [filteredCards.length]);
   439→
   440→  const handleAddCard = useCallback(() => {
   441→    if (!newFront.trim() || !newBack.trim()) {
   442→      toast('لطفاً هر دو طرف کارت رو پر کن');
   443→      return;
   444→    }
   445→    addFlashcard({
   446→      id: Date.now().toString(),
   447→      front: newFront.trim(),
   448→      back: newBack.trim(),
   449→      isSystem: false,
   450→      mastery: 'review',
   451→      subject: newSubject || undefined,
   452→    });
   453→    setNewFront('');
   454→    setNewBack('');
   455→    setNewSubject('');
   456→    setShowAddForm(false);
   457→    toast('کارت جدید اضافه شد! 🎉');
   458→  }, [newFront, newBack, newSubject, addFlashcard]);
   459→
   460→  // Count cards per mastery
   461→  const masteredCount = flashcards.filter((c) => c.mastery === 'mastered').length;
   462→  const reviewCount = flashcards.filter((c) => c.mastery === 'review').length;
   463→  const weakCount = flashcards.filter((c) => c.mastery === 'weak').length;
   464→
   465→  return (
   466→    <div>
   467→      {/* Main Tabs: Study / Marked */}
   468→      <div className="flex gap-2 mb-4">
   469→        <motion.button
   470→          onClick={() => { setTab('study'); setMasteryFilter('all'); setCurrentIndex(0); setIsFlipped(false); }}
   471→          whileHover={{ scale: 1.02 }}
   472→          whileTap={{ scale: 0.95 }}
   473→          className={`flex-1 py-2.5 rounded-xl text-sm font-medium transition-colors min-h-[44px] ${
   474→            tab === 'study'
   475→              ? 'bg-[#3EB489]/20 text-[#3EB489] border border-[#3EB489]/30'
   476→              : 'bg-zinc-900/50 text-zinc-400 border border-white/5'
   477→          }`}
   478→        >
   479→          📖 مطالعه
   480→        </motion.button>
   481→        <motion.button
   482→          onClick={() => { setTab('marked'); setMasteryFilter('all'); setCurrentIndex(0); setIsFlipped(false); }}
   483→          whileHover={{ scale: 1.02 }}
   484→          whileTap={{ scale: 0.95 }}
   485→          className={`flex-1 py-2.5 rounded-xl text-sm font-medium transition-colors min-h-[44px] relative ${
   486→            tab === 'marked'
   487→              ? 'bg-[#3EB489]/20 text-[#3EB489] border border-[#3EB489]/30'
   488→              : 'bg-zinc-900/50 text-zinc-400 border border-white/5'
   489→          }`}
   490→        >
   491→          🏷️ نشانه‌گذاری شده
   492→          {(reviewCount + weakCount) > 0 && (
   493→            <span className="absolute -top-1 -left-1 w-5 h-5 bg-amber-500 rounded-full text-[10px] font-bold text-zinc-950 flex items-center justify-center">
   494→              {reviewCount + weakCount}
   495→            </span>
   496→          )}
   497→        </motion.button>
   498→      </div>
   499→
   500→      {/* Subject Selector */}
   501→      <div className="mb-3">
   502→        <p className="text-xs text-zinc-500 mb-2 px-1">انتخاب درس</p>
   503→        <div className="flex gap-2 overflow-x-auto pb-1 -mx-1 px-1" style={{ scrollbarWidth: 'none' }}>
   504→          <motion.button
   505→            onClick={() => { setSelectedSubject(null); setCurrentIndex(0); setIsFlipped(false); }}
   506→            whileHover={{ scale: 1.02 }}
   507→            whileTap={{ scale: 0.95 }}
   508→            className={`shrink-0 px-3 py-1.5 rounded-lg text-xs font-medium transition-colors ${
   509→              !selectedSubject
   510→                ? 'bg-zinc-200 text-zinc-950'
   511→                : 'bg-zinc-900/50 text-zinc-400 border border-white/5'
   512→            }`}
   513→          >
   514→            همه
   515→          </motion.button>
   516→          {allSubjects.map((subject) => (

   982→// ===== Tool 4: Percentage Calculator =====
   983→function CalculatorTool() {
   984→  const [correct, setCorrect] = useState('');
   985→  const [unanswered, setUnanswered] = useState('');
   986→  const [incorrect, setIncorrect] = useState('');
   987→
   988→  const correctNum = parseInt(correct) || 0;
   989→  const unansweredNum = parseInt(unanswered) || 0;
   990→  const incorrectNum = parseInt(incorrect) || 0;
   991→
   992→  // Auto-calculate total
   993→  const totalNum = correctNum + unansweredNum + incorrectNum;
   994→
   995→  // Main percentage: ((Correct × 3) - Incorrect) / (Total × 3) × 100
   996→  const denominator = totalNum * 3;
   997→  const numerator = correctNum * 3 - incorrectNum;
   998→  const result = denominator > 0 ? (numerator / denominator) * 100 : 0;
   999→  const isNegative = result < 0;
  1000→
  1001→  // What-if: If you left wrong answers unanswered instead
  1002→  // Total would be (correct + unanswered) instead of (correct + unanswered + incorrect)
  1003→  // Score would be (correct × 3) / ((correct + unanswered) × 3) × 100 = correct / (correct + unanswered) × 100
  1004→  const whatIfTotal = correctNum + unansweredNum;
  1005→  const whatIfResult = whatIfTotal > 0 ? (correctNum / whatIfTotal) * 100 : 0;
  1006→  const whatIfDiff = whatIfTotal > 0 ? whatIfResult - result : 0;
  1007→
  1008→  const resultColor = isNegative
  1009→    ? 'text-rose-500'
  1010→    : result >= 50

  1153→    </div>

  1160→  const [phase, setPhase] = useState<'inhale' | 'hold' | 'exhale'>('inhale');
  1161→  const [cycleCount, setCycleCount] = useState(0);

  1165→    inhale: { text: 'دم', subtext: 'بکش...', duration: 4000, scale: 1 },

  1193→    return () => {

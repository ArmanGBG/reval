    29→function toPersianDigits(num: number | string): string {
    30→  const persianDigits = ['۰', '۱', '۲', '۳', '۴', '۵', '۶', '۷', '۸', '۹'];
    31→  return num
    32→    .toString()
    33→    .split('')
    34→    .map((d) => persianDigits[parseInt(d)] ?? d)
    35→    .join('');
    36→}

   253→// ============================================================
   254→// Chapter Row (with expandable topics)

   256→function ChapterRow({
   257→  subject,
   258→  chapter,
   259→  depth,
   260→  onEdit,
   261→  onAddTopic,
   262→  onEditTopic,
   263→  onRefresh,
   264→}: {
   265→  subject: Subject;
   266→  chapter: Chapter;
   267→  depth: number;
   268→  onEdit: () => void;
   269→  onAddTopic: () => void;
   270→  onEditTopic: (topic: Topic) => void;
   271→  onRefresh: () => void;
   272→}) {
   273→  const [expanded, setExpanded] = useState(false);
   274→  const [deleting, setDeleting] = useState(false);
   275→  const hasTopics = depth === 3;
   276→  const topicCount = chapter.topics?.length || 0;
   277→
   278→  const handleDelete = async () => {
   279→    if (!confirm(`فصل «${chapter.title}» حذف شود؟`)) return;
   280→    setDeleting(true);
   281→    try {
   282→      const res = await fetch(
   283→        `/api/subjects/${subject.id}/chapters/${chapter.id}`,
   284→        { method: 'DELETE' }
   285→      );
   286→      if (!res.ok) {
   287→        const data = await res.json();
   288→        throw new Error(data.error || 'خطا در حذف');
   289→      }
   290→      toast.success('فصل حذف شد');
   291→      onRefresh();
   292→    } catch (err) {
   293→      const msg = err instanceof Error ? err.message : 'خطا در حذف فصل';
   294→      toast.error(msg);
   295→    } finally {
   296→      setDeleting(false);
   297→    }
   298→  };
   299→
   300→  return (
   301→    <div className="bg-[var(--bg-elevated)] rounded-xl border border-[var(--border)] overflow-hidden">
   302→      <div className="flex items-center gap-2 p-3">
   303→        {/* Expand toggle (only for depth 3) */}
   304→        {hasTopics ? (
   305→          <button
   306→            onClick={() => setExpanded(!expanded)}
   307→            className="icon-btn w-7 h-7 rounded-lg flex items-center justify-center text-[var(--foreground-muted)]"
   308→          >
   309→            {expanded ? (
   310→              <ChevronDown className="w-3.5 h-3.5" />
   311→            ) : (
   312→              <ChevronLeft className="w-3.5 h-3.5" />
   313→            )}
   314→          </button>
   315→        ) : (
   316→          <div className="w-7 h-7 flex items-center justify-center">
   317→            <Layers className="w-3.5 h-3.5 text-[var(--foreground-subtle)]" />
   318→          </div>
   319→        )}
   320→
   321→        {/* Chapter number badge */}
   322→        <span className="text-[10px] font-bold text-[var(--foreground-subtle)] w-6 text-center">
   323→          {toPersianDigits(chapter.chapterNo)}
   324→        </span>
   325→
   326→        {/* Title */}
   327→        <button
   328→          onClick={() => hasTopics && setExpanded(!expanded)}
   329→          className="flex-1 text-right text-sm font-medium text-[var(--foreground)] hover:text-[var(--gold)] transition-colors"
   330→        >
   331→          {chapter.title}
   332→        </button>
   333→
   334→        {/* Topic count (depth 3) */}
   335→        {hasTopics && (
   336→          <span className="text-[10px] text-[var(--foreground-muted)]">
   337→            {toPersianDigits(topicCount)} گفتار
   338→          </span>
   339→        )}
   340→
   341→        {/* Actions */}
   342→        <button
   343→          onClick={onEdit}
   344→          className="icon-btn w-7 h-7 rounded-lg flex items-center justify-center text-[var(--foreground-muted)]"
   345→          aria-label="ویرایش فصل"
   346→        >
   347→          <Pencil className="w-3 h-3" />
   348→        </button>
   349→        <button
   350→          onClick={handleDelete}
   351→          disabled={deleting}
   352→          className="icon-btn w-7 h-7 rounded-lg flex items-center justify-center text-[var(--foreground-muted)] hover:text-rose-400 disabled:opacity-50"
   353→          aria-label="حذف فصل"
   354→        >
   355→          {deleting ? (
   356→            <Loader2 className="w-3 h-3 animate-spin" />
   357→          ) : (
   358→            <Trash2 className="w-3 h-3" />
   359→          )}
   360→        </button>
   361→      </div>
   362→
   363→      {/* Topics list (depth 3) */}
   364→      <AnimatePresence>
   365→        {expanded && hasTopics && (
   366→          <motion.div
   367→            initial={{ height: 0, opacity: 0 }}
   368→            animate={{ height: 'auto', opacity: 1 }}
   369→            exit={{ height: 0, opacity: 0 }}
   370→            transition={{ duration: 0.2 }}
   371→            className="border-t border-[var(--border)] overflow-hidden"
   372→          >
   373→            <div className="p-3 space-y-1.5 bg-[var(--bg-deep)]/40">
   374→              {chapter.topics && chapter.topics.length > 0 ? (
   375→                chapter.topics.map((topic) => (
   376→                  <div
   377→                    key={topic.id}
   378→                    className="flex items-center gap-2 p-2 rounded-lg hover:bg-[var(--bg-elevated)] nav-item-hover"
   379→                  >
   380→                    <MessageSquare className="w-3 h-3 text-[var(--foreground-subtle)]" />
   381→                    <span className="text-[10px] text-[var(--foreground-subtle)] w-5">
   382→                      {toPersianDigits(topic.topicNo)}
   383→                    </span>
   384→                    <span className="flex-1 text-xs text-[var(--foreground-muted)]">
   385→                      {topic.title}
   386→                    </span>
   387→                    <button
   388→                      onClick={() => onEditTopic(topic)}
   389→                      className="icon-btn w-6 h-6 rounded flex items-center justify-center text-[var(--foreground-subtle)]"
   390→                    >
   391→                      <Pencil className="w-2.5 h-2.5" />
   392→                    </button>
   393→                  </div>
   394→                ))
   395→              ) : (
   396→                <p className="text-xs text-[var(--foreground-subtle)] text-center py-2">
   397→                  گفتاری ثبت نشده
   398→                </p>
   399→              )}
   400→              <button
   401→                onClick={onAddTopic}
   402→                className="btn-hover w-full h-8 rounded-lg border border-dashed border-[var(--border)] text-[var(--foreground-muted)] hover:text-[var(--gold)] hover:border-[var(--gold)]/40 text-[11px] font-medium flex items-center justify-center gap-1.5"
   403→              >
   404→                <Plus className="w-3 h-3" />
   405→                افزودن گفتار
   406→              </button>
   407→            </div>
   408→          </motion.div>
   409→        )}
   410→      </AnimatePresence>
   411→    </div>
   412→  );
   413→}

   415→// ============================================================
   416→// Add Grade Modal
   417→// ============================================================
   418→function AddGradeModal({
   419→  subjectId,
   420→  availableGrades,
   421→  onClose,
   422→  onSaved,
   423→}: {
   424→  subjectId: string;
   425→  availableGrades: string[];
   426→  onClose: () => void;
   427→  onSaved: () => void;
   428→}) {
   429→  const [grade, setGrade] = useState(availableGrades[0] || '');
   430→  const [major, setMajor] = useState<string>('تجربی');
   431→  const [depth, setDepth] = useState(2);
   432→  const [allowOptionalSubtopic, setAllowOptionalSubtopic] = useState(false);
   433→  const [saving, setSaving] = useState(false);
   434→
   435→  const handleSubmit = async () => {
   436→    if (!grade || !major) {
   437→      toast.error('پایه و رشته الزامی است');
   438→      return;
   439→    }
   440→    setSaving(true);
   441→    try {
   442→      const res = await fetch(`/api/subjects/${subjectId}/grades`, {
   443→        method: 'POST',
   444→        headers: { 'Content-Type': 'application/json' },
   445→        body: JSON.stringify({ grade, major, depth, allowOptionalSubtopic }),
   446→      });
   447→      const data = await res.json();
   448→      if (!res.ok) throw new Error(data.error || 'خطا در ایجاد پایه');
   449→      onSaved();
   450→    } catch (err) {
   451→      const msg = err instanceof Error ? err.message : 'خطا در ایجاد پایه';
   452→      toast.error(msg);
   453→    } finally {
   454→      setSaving(false);
   455→    }
   456→  };
   457→
   458→  return (
   459→    <motion.div
   460→      initial={{ opacity: 0 }}
   461→      animate={{ opacity: 1 }}
   462→      exit={{ opacity: 0 }}
   463→      className="fixed inset-0 z-[80] flex items-center justify-center p-4 bg-black/70 backdrop-blur-sm"
   464→      onClick={onClose}
   465→    >
   466→      <motion.div
   467→        initial={{ opacity: 0, scale: 0.96, y: 16 }}
   468→        animate={{ opacity: 1, scale: 1, y: 0 }}
   469→        exit={{ opacity: 0, scale: 0.96, y: 16 }}
   470→        onClick={(e) => e.stopPropagation()}
   471→        className="surface-2 edge-highlight rounded-3xl w-full max-w-md p-6"
   472→      >
   473→        <div className="flex items-center justify-between mb-5">
   474→          <h2 className="text-lg font-bold text-[var(--foreground)]">افزودن پایه جدید</h2>
   475→          <button
   476→            onClick={onClose}
   477→            className="icon-btn w-9 h-9 rounded-xl border border-[var(--border)] flex items-center justify-center text-[var(--foreground-muted)]"
   478→          >
   479→            <X className="w-4 h-4" />
   480→          </button>
   481→        </div>
   482→
   483→        {/* Grade selector */}
   484→        <div className="mb-4">
   485→          <label className="text-xs font-medium text-[var(--foreground-muted)] mb-1.5 block">
   486→            پایه
   487→          </label>
   488→          <div className="grid grid-cols-2 gap-2">
   489→            {availableGrades.map((g) => (
   490→              <button
   491→                key={g}
   492→                onClick={() => setGrade(g)}
   493→                className={`btn-hover h-10 rounded-xl text-xs font-medium border ${
   494→                  grade === g
   495→                    ? 'bg-[var(--gold-soft)] border-[var(--gold)]/40 text-[var(--gold)]'
   496→                    : 'bg-[var(--bg-elevated)] border-[var(--border)] text-[var(--foreground-muted)]'
   497→                }`}
   498→              >
   499→                {g}
   500→              </button>
   501→            ))}
   502→          </div>
   503→        </div>
   504→
   505→        {/* Major selector */}
   506→        <div className="mb-4">
   507→          <label className="text-xs font-medium text-[var(--foreground-muted)] mb-1.5 block">
   508→            رشته
   509→          </label>
   510→          <div className="grid grid-cols-3 gap-2">
   511→            {(MAJORS as readonly string[]).map((m) => (
   512→              <button
   513→                key={m}
   514→                onClick={() => setMajor(m)}
   515→                className={`btn-hover h-10 rounded-xl text-xs font-medium border ${
   516→                  major === m
   517→                    ? 'bg-[var(--gold-soft)] border-[var(--gold)]/40 text-[var(--gold)]'

   527→        {/* Depth selector */}

   529→          <label className="text-xs font-medium text-[var(--foreground-muted)] mb-1.5 block">
   530→            تعداد لایه‌های درخت
   531→          </label>

   536→                onClick={() => setDepth(opt.value)}

   538→                  depth === opt.value

   541→                }`}

   575→        </div>

   595→            انصراف
   596→          </button>
   597→        </div>
   598→      </motion.div>
   599→    </motion.div>

   604→// Chapter Form Modal (Add/Edit)
   605→// ============================================================
   606→function ChapterFormModal({
   607→  subjectId,
   608→  grade,
   609→  chapter,
   610→  onClose,
   611→  onSaved,
   612→}: {
   613→  subjectId: string;
   614→  grade: string;
   615→  chapter: Chapter | null;
   616→  onClose: () => void;
   617→  onSaved: () => void;
   618→}) {
   619→  const [title, setTitle] = useState(chapter?.title || '');
   620→  const [saving, setSaving] = useState(false);
   621→  const isEdit = !!chapter;
   622→
   623→  const handleSubmit = async () => {
   624→    if (!title.trim()) {
   625→      toast.error('عنوان فصل الزامی است');
   626→      return;
   627→    }
   628→    setSaving(true);
   629→    try {
   630→      const url = isEdit
   631→        ? `/api/subjects/${subjectId}/chapters/${chapter!.id}`
   632→        : `/api/subjects/${subjectId}/chapters`;
   633→      const method = isEdit ? 'PATCH' : 'POST';
   634→      const res = await fetch(url, {
   635→        method,
   636→        headers: { 'Content-Type': 'application/json' },
   637→        body: JSON.stringify({ grade, title: title.trim() }),
   638→      });
   639→      const data = await res.json();
   640→      if (!res.ok) throw new Error(data.error || 'خطا');
   641→      onSaved();
   642→    } catch (err) {
   643→      const msg = err instanceof Error ? err.message : 'خطا';
   644→      toast.error(msg);
   645→    } finally {
   646→      setSaving(false);
   647→    }
   648→  };
   649→
   650→  return (
   651→    <motion.div
   652→      initial={{ opacity: 0 }}
   653→      animate={{ opacity: 1 }}
   654→      exit={{ opacity: 0 }}
   655→      className="fixed inset-0 z-[80] flex items-center justify-center p-4 bg-black/70 backdrop-blur-sm"
   656→      onClick={onClose}
   657→    >
   658→      <motion.div
   659→        initial={{ opacity: 0, scale: 0.96, y: 16 }}
   660→        animate={{ opacity: 1, scale: 1, y: 0 }}
   661→        exit={{ opacity: 0, scale: 0.96, y: 16 }}
   662→        onClick={(e) => e.stopPropagation()}
   663→        className="surface-2 edge-highlight rounded-3xl w-full max-w-md p-6"
   664→      >
   665→        <div className="flex items-center justify-between mb-5">
   666→          <h2 className="text-lg font-bold text-[var(--foreground)]">
   667→            {isEdit ? 'ویرایش فصل' : `افزودن فصل به پایه ${grade}`}
   668→          </h2>
   669→          <button
   670→            onClick={onClose}
   671→            className="icon-btn w-9 h-9 rounded-xl border border-[var(--border)] flex items-center justify-center text-[var(--foreground-muted)]"
   672→          >
   673→            <X className="w-4 h-4" />
   674→          </button>
   675→        </div>
   676→
   677→        <div className="mb-4">
   678→          <label className="text-xs font-medium text-[var(--foreground-muted)] mb-1.5 block">
   679→            عنوان فصل
   680→          </label>
   681→          <input
   682→            type="text"
   683→            value={title}
   684→            onChange={(e) => setTitle(e.target.value)}
   685→            placeholder="مثلاً: فصل ۱: دنیای زنده"
   686→            autoFocus
   687→            className="w-full bg-[var(--bg-elevated)] border border-[var(--border)] rounded-xl px-3 h-11 text-sm text-[var(--foreground)] placeholder:text-[var(--foreground-subtle)] focus:outline-none focus:border-[var(--gold)]/40"
   688→          />
   689→        </div>
   690→
   691→        <div className="flex gap-2 pt-4 border-t border-[var(--border)]">
   692→          <button
   693→            onClick={handleSubmit}
   694→            disabled={saving || !title.trim()}
   695→            className="btn-hover glow-hover-gold glow-hover flex-1 h-11 rounded-xl bg-[var(--gold)] text-zinc-950 font-bold text-sm disabled:opacity-50 flex items-center justify-center gap-2"
   696→          >
   697→            {saving ? (
   698→              <Loader2 className="w-4 h-4 animate-spin" />
   699→            ) : (
   700→              <>
   701→                <Save className="w-4 h-4" />
   702→                {isEdit ? 'ذخیره تغییرات' : 'افزودن فصل'}
   703→              </>
   704→            )}
   705→          </button>
   706→          <button
   707→            onClick={onClose}
   708→            className="btn-hover h-11 px-5 rounded-xl border border-[var(--border)] text-[var(--foreground-muted)] text-sm"
   709→          >
   710→            انصراف
   711→          </button>
   712→        </div>
   713→      </motion.div>
   714→    </motion.div>
   715→  );
   716→}

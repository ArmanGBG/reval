    65→    65→function toPersianDigits(num: number | string): string {
    66→    66→  const persianDigits = ['۰', '۱', '۲', '۳', '۴', '۵', '۶', '۷', '۸', '۹'];
    67→    67→  return num.toString().split('').map((d) => persianDigits[parseInt(d)] ?? d).join('');
    68→    68→}

    71→    71→function computeStudentStatus(student: StudentProfile): StudentStatus {
    72→    72→  const studyRatio = student.studyHoursPerWeek / student.studyHoursTarget;
    73→    73→  if (student.mockExamScore >= 75 && studyRatio >= 0.85 && student.taskCompletionRate >= 75 && student.mood !== 'poor' && student.mood !== 'critical') return 'excellent';
    74→    74→  if (student.mockExamScore < 40 || student.mood === 'critical' || student.attendanceRate < 60 || student.taskCompletionRate < 30) return 'critical';
    75→    75→  if (student.mockExamScore < 55 || student.mood === 'poor' || (student.mockExamScore - student.previousMockScore < -8 && student.studyHoursTrend === 'down') || student.attendanceRate < 70) return 'at-risk';
    76→    76→  if (student.mockExamScore >= 55 && student.taskCompletionRate >= 55 && studyRatio >= 0.65) return 'good';
    77→    77→  return 'fair';
    78→    78→}

    80→    80→function computeRisks(students: StudentProfile[]): StudentRisk[] {
    81→    81→  return students.map((student) => {
    82→    82→    const reasons: string[] = [];
    83→    83→    let level: 'high' | 'medium' | 'low' = 'low';
    84→    84→    const scoreDiff = student.mockExamScore - student.previousMockScore;
    85→    85→
    86→    86→    if (student.mockExamScore < 40) { reasons.push(`نمره آزمون بسیار پایین (${toPersianDigits(student.mockExamScore)})`); level = 'high'; }
    87→    87→    if (scoreDiff < -8) { reasons.push(`افت شدید نمره (${toPersianDigits(Math.abs(scoreDiff))} نمره کاهش)`); level = 'high'; }
    88→    88→    if (student.studyHoursTrend === 'down' && student.studyHoursPerWeek < student.studyHoursTarget * 0.7) { reasons.push(`کاهش ساعت مطالعه (${toPersianDigits(student.studyHoursPerWeek)} ساعت از ${toPersianDigits(student.studyHoursTarget)})`); level = level === 'high' ? 'high' : 'medium'; }
    89→    89→    if (student.mood === 'critical' || student.mood === 'poor') { reasons.push(`وضعیت روحی ${student.mood === 'critical' ? 'بحرانی' : 'ضعیف'}`); level = 'high'; }
    90→    90→    if (student.attendanceRate < 65) { reasons.push(`حضور کم (${toPersianDigits(student.attendanceRate)}٪)`); level = level === 'high' ? 'high' : 'medium'; }
    91→    91→    if (student.taskCompletionRate < 35) { reasons.push(`تکمیل وظایف پایین (${toPersianDigits(student.taskCompletionRate)}٪)`); level = level === 'high' ? 'high' : 'medium'; }
    92→    92→    if (student.stressLevel >= 8) { reasons.push(`استرس بالا (${toPersianDigits(student.stressLevel)}/۱۰)`); level = level === 'high' ? 'high' : 'medium'; }
    93→    93→    if (student.motivationLevel <= 3) { reasons.push(`انگیزه پایین (${toPersianDigits(student.motivationLevel)}/۱۰)`); level = level === 'high' ? 'high' : 'medium'; }
    94→    94→
    95→    95→    let immediateAction = '';
    96→    96→    if (level === 'high') {
    97→    97→      if (student.mood === 'critical' || student.mood === 'poor') immediateAction = 'جلسه فوری با روانشناس + بازنگری برنامه';
    98→    98→      else if (student.attendanceRate < 65) immediateAction = 'تماس با خانواده + بررسی موانع حضور';
    99→    99→      else immediateAction = 'جلسه اضطراری + تمرکز بر ۲ درس ضعیف';
   100→   100→    } else if (level === 'medium') {
   101→   101→      immediateAction = 'جلسه هفتگی + هدف‌گذاری خرد';
   102→   102→    } else {
   103→   103→      immediateAction = 'ادامه برنامه + تشویق و تثبیت';
   104→   104→    }
   105→   105→
   106→   106→    return { studentId: student.id, level, reasons, immediateAction };
   107→   107→  });
   108→   108→}

   149→   149→    if (student.flashcardsMastered < student.flashcardsTotal * 0.4) interventions.push(`${toPersianDigits(15)} دقیقه فلش‌کارت قبل از خواب - دروس ضعیف`);

   152→   152→
   153→   153→    return { studentId: student.id, strengths, weaknesses, psychologicalAssessment, interventions };
   154→   154→  });
   155→   155→}

   208→   208→function MiniRadar({ grades }: { grades: Record<string, number> }) {
   209→   209→  const entries = Object.entries(grades);
   210→   210→  const size = 120;
   211→   211→  const cx = size / 2;
   212→   212→  const cy = size / 2;
   213→   213→  const r = size / 2 - 15;
   214→   214→  const points = entries.map(([_, val], i) => {
   215→   215→    const angle = (Math.PI * 2 * i) / entries.length - Math.PI / 2;
   216→   216→    return { x: cx + r * (val / 20) * Math.cos(angle), y: cy + r * (val / 20) * Math.sin(angle) };
   217→   217→  });
   218→   218→  const pathData = points.map((p, i) => `${i === 0 ? 'M' : 'L'} ${p.x} ${p.y}`).join(' ') + ' Z';
   219→   219→
   220→   220→  return (
   221→   221→    <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} className="mx-auto">
   222→   222→      {[0.25, 0.5, 0.75, 1].map((s) => <circle key={s} cx={cx} cy={cy} r={r * s} fill="none" stroke="rgba(255,255,255,0.06)" strokeWidth={1} />)}
   223→   223→      {entries.map(([_, _v], i) => { const a = (Math.PI * 2 * i) / entries.length - Math.PI / 2; return <line key={i} x1={cx} y1={cy} x2={cx + r * Math.cos(a)} y2={cy + r * Math.sin(a)} stroke="rgba(255,255,255,0.06)" strokeWidth={1} />; })}
   224→   224→      <path d={pathData} fill="rgba(62,180,137,0.15)" stroke="#3EB489" strokeWidth={1.5} />
   225→   225→      {points.map((p, i) => <circle key={i} cx={p.x} cy={p.y} r={3} fill="#3EB489" />)}
   226→   226→      {entries.map(([name], i) => { const a = (Math.PI * 2 * i) / entries.length - Math.PI / 2; return <text key={i} x={cx + (r + 12) * Math.cos(a)} y={cy + (r + 12) * Math.sin(a)} textAnchor="middle" dominantBaseline="middle" className="fill-zinc-400 text-[7px]" style={{ fontFamily: 'var(--font-vazirmatn)' }}>{name}</text>; })}
   227→   227→    </svg>
   228→   228→  );
   229→   229→}

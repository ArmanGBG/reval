    65→function toPersianDigits(num: number | string): string {
    66→  const persianDigits = ['۰', '۱', '۲', '۳', '۴', '۵', '۶', '۷', '۸', '۹'];
    67→  return num.toString().split('').map((d) => persianDigits[parseInt(d)] ?? d).join('');
    68→}

    70→// ===== Analysis Engine =====
    71→function computeStudentStatus(student: StudentProfile): StudentStatus {
    72→  const studyRatio = student.studyHoursPerWeek / student.studyHoursTarget;
    73→  if (student.mockExamScore >= 75 && studyRatio >= 0.85 && student.taskCompletionRate >= 75 && student.mood !== 'poor' && student.mood !== 'critical') return 'excellent';
    74→  if (student.mockExamScore < 40 || student.mood === 'critical' || student.attendanceRate < 60 || student.taskCompletionRate < 30) return 'critical';
    75→  if (student.mockExamScore < 55 || student.mood === 'poor' || (student.mockExamScore - student.previousMockScore < -8 && student.studyHoursTrend === 'down') || student.attendanceRate < 70) return 'at-risk';
    76→  if (student.mockExamScore >= 55 && student.taskCompletionRate >= 55 && studyRatio >= 0.65) return 'good';
    77→  return 'fair';
    78→}

    80→function computeRisks(students: StudentProfile[]): StudentRisk[] {
    81→  return students.map((student) => {
    82→    const reasons: string[] = [];
    83→    let level: 'high' | 'medium' | 'low' = 'low';
    84→    const scoreDiff = student.mockExamScore - student.previousMockScore;
    85→
    86→    if (student.mockExamScore < 40) { reasons.push(`نمره آزمون بسیار پایین (${toPersianDigits(student.mockExamScore)})`); level = 'high'; }
    87→    if (scoreDiff < -8) { reasons.push(`افت شدید نمره (${toPersianDigits(Math.abs(scoreDiff))} نمره کاهش)`); level = 'high'; }
    88→    if (student.studyHoursTrend === 'down' && student.studyHoursPerWeek < student.studyHoursTarget * 0.7) { reasons.push(`کاهش ساعت مطالعه (${toPersianDigits(student.studyHoursPerWeek)} ساعت از ${toPersianDigits(student.studyHoursTarget)})`); level = level === 'high' ? 'high' : 'medium'; }
    89→    if (student.mood === 'critical' || student.mood === 'poor') { reasons.push(`وضعیت روحی ${student.mood === 'critical' ? 'بحرانی' : 'ضعیف'}`); level = 'high'; }
    90→    if (student.attendanceRate < 65) { reasons.push(`حضور کم (${toPersianDigits(student.attendanceRate)}٪)`); level = level === 'high' ? 'high' : 'medium'; }
    91→    if (student.taskCompletionRate < 35) { reasons.push(`تکمیل وظایف پایین (${toPersianDigits(student.taskCompletionRate)}٪)`); level = level === 'high' ? 'high' : 'medium'; }
    92→    if (student.stressLevel >= 8) { reasons.push(`استرس بالا (${toPersianDigits(student.stressLevel)}/۱۰)`); level = level === 'high' ? 'high' : 'medium'; }
    93→    if (student.motivationLevel <= 3) { reasons.push(`انگیزه پایین (${toPersianDigits(student.motivationLevel)}/۱۰)`); level = level === 'high' ? 'high' : 'medium'; }
    94→
    95→    let immediateAction = '';
    96→    if (level === 'high') {
    97→      if (student.mood === 'critical' || student.mood === 'poor') immediateAction = 'جلسه فوری با روانشناس + بازنگری برنامه';
    98→      else if (student.attendanceRate < 65) immediateAction = 'تماس با خانواده + بررسی موانع حضور';
    99→      else immediateAction = 'جلسه اضطراری + تمرکز بر ۲ درس ضعیف';
   100→    } else if (level === 'medium') {
   101→      immediateAction = 'جلسه هفتگی + هدف‌گذاری خرد';
   102→    } else {
   103→      immediateAction = 'ادامه برنامه + تشویق و تثبیت';
   104→    }
   105→
   106→    return { studentId: student.id, level, reasons, immediateAction };
   107→  });
   108→}

   110→function computeAnalyses(students: StudentProfile[]): StudentAnalysis[] {
   111→  return students.map((student) => {
   112→    const strengths: string[] = [];
   113→    const weaknesses: string[] = [];
   114→    const interventions: string[] = [];
   115→
   116→    if (student.mockExamScore >= 70) strengths.push(`نمره آزمون خوب (${toPersianDigits(student.mockExamScore)})`);
   117→    if (student.studyHoursTrend === 'up') strengths.push('روند افزایشی ساعت مطالعه');
   118→    if (student.attendanceRate >= 90) strengths.push(`حضور منظم (${toPersianDigits(student.attendanceRate)}٪)`);
   119→    if (student.taskCompletionRate >= 80) strengths.push(`تکمیل بالا (${toPersianDigits(student.taskCompletionRate)}٪)`);
   120→    if (student.motivationLevel >= 7) strengths.push(`انگیزه بالا (${toPersianDigits(student.motivationLevel)}/۱۰)`);
   121→    if (student.pomodoroSessionsPerWeek >= 15) strengths.push('استفاده منظم از پومودورو');
   122→    const strongSubjects = Object.entries(student.schoolGrades).filter(([_, g]) => g >= 17).map(([n]) => n);
   123→    if (strongSubjects.length > 0) strengths.push(`قوی در: ${strongSubjects.join('، ')}`);
   124→
   125→    if (student.mockExamScore < 55) weaknesses.push(`نمره پایین (${toPersianDigits(student.mockExamScore)})`);
   126→    if (student.studyHoursTrend === 'down') weaknesses.push('کاهش ساعت مطالعه');
   127→    if (student.attendanceRate < 75) weaknesses.push(`حضور ناکافی (${toPersianDigits(student.attendanceRate)}٪)`);
   128→    if (student.taskCompletionRate < 50) weaknesses.push(`تکمیل پایین (${toPersianDigits(student.taskCompletionRate)}٪)`);
   129→    if (student.stressLevel >= 7) weaknesses.push(`استرس بالا (${toPersianDigits(student.stressLevel)}/۱۰)`);
   130→    const weakSubjects = Object.entries(student.schoolGrades).filter(([_, g]) => g < 13).map(([n]) => n);
   131→    if (weakSubjects.length > 0) weaknesses.push(`ضعیف در: ${weakSubjects.join('، ')}`);
   132→    const deficit = student.studyHoursTarget - student.studyHoursPerWeek;
   133→    if (deficit > 10) weaknesses.push(`کمبود ${toPersianDigits(deficit)} ساعت مطالعه`);
   134→
   135→    let psychologicalAssessment = '';
   136→    if (student.mood === 'critical') psychologicalAssessment = 'وضعیت بحرانی. نیاز به ارجاع فوری به روانشناس. علائم فرسودگی و احتمال افسردگی.';
   137→    else if (student.mood === 'poor') psychologicalAssessment = 'وضعیت ضعیف. نشانه‌های فرسودگی تحصیلی. نیاز به کاهش فشار و تمرکز بر سلامت روان.';
   138→    else if (student.stressLevel >= 7 && student.motivationLevel <= 5) psychologicalAssessment = 'استرس بالا + انگیزه پایین - الگوی خطرناک. احتمال شروع فرسودگی.';
   139→    else if (student.stressLevel >= 7) psychologicalAssessment = 'استرس بالا اما انگیزه حفظ شده. در معرض خطر فرسودگی.';
   140→    else if (student.motivationLevel >= 7 && student.stressLevel <= 5) psychologicalAssessment = 'تعادل خوب بین انگیزه و استرس. ادامه روند فعلی توصیه می‌شود.';
   141→    else psychologicalAssessment = 'وضعیت متوسط. نیاز به پیگیری و تقویت انگیزه.';
   142→
   143→    if (weakSubjects.length > 0) interventions.push(`تخصیص ${toPersianDigits(weakSubjects.length * 2)} ساعت اضافی برای ${weakSubjects.join(' و ')} - مطالعه صبحگاهی`);
   144→    if (student.studyHoursTrend === 'down') interventions.push(`بازنگری برنامه: کاهش حجم و افزایش کیفیت - تکنیک پومودورو`);
   145→    if (student.taskCompletionRate < 50) interventions.push(`شروع با وظایف کوچک‌تر: تقسیم هر وظیفه به ${toPersianDigits(3)} بخش`);
   146→    if (student.stressLevel >= 7) interventions.push('استفاده روزانه از اورژانس استرس + تمرین ذهن‌آگاهی ۱۰ دقیقه');
   147→    if (student.attendanceRate < 75) interventions.push('برنامه انعطاف‌پذیر: امکان جبران آنلاین + جلسه هفتگی');
   148→    if (student.motivationLevel <= 4) interventions.push('هدف‌گذاری خرد: اهداف روزانه کوچک + پاداش هفتگی');
   149→    if (student.flashcardsMastered < student.flashcardsTotal * 0.4) interventions.push(`${toPersianDigits(15)} دقیقه فلش‌کارت قبل از خواب - دروس ضعیف`);
   150→    if (student.mockExamScore >= 75 && student.taskCompletionRate >= 80) interventions.push('ادامه روند + تست‌های سنجشی سخت‌تر + مدیریت زمان آزمون');
   151→    if (student.weeksUntilExam <= 16 && student.mockExamScore < 60) interventions.push('تمرکز بر تست‌زنی دروس با بالاترین ضریب + مرور سریع');
   152→
   153→    return { studentId: student.id, strengths, weaknesses, psychologicalAssessment, interventions };
   154→  });
   155→}

   208→function MiniRadar({ grades }: { grades: Record<string, number> }) {
   209→  const entries = Object.entries(grades);
   210→  const size = 120;
   211→  const cx = size / 2;
   212→  const cy = size / 2;
   213→  const r = size / 2 - 15;
   214→  const points = entries.map(([_, val], i) => {
   215→    const angle = (Math.PI * 2 * i) / entries.length - Math.PI / 2;
   216→    return { x: cx + r * (val / 20) * Math.cos(angle), y: cy + r * (val / 20) * Math.sin(angle) };
   217→  });
   218→  const pathData = points.map((p, i) => `${i === 0 ? 'M' : 'L'} ${p.x} ${p.y}`).join(' ') + ' Z';
   219→
   220→  return (
   221→    <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} className="mx-auto">
   222→      {[0.25, 0.5, 0.75, 1].map((s) => <circle key={s} cx={cx} cy={cy} r={r * s} fill="none" stroke="rgba(255,255,255,0.06)" strokeWidth={1} />)}
   223→      {entries.map(([_, _v], i) => { const a = (Math.PI * 2 * i) / entries.length - Math.PI / 2; return <line key={i} x1={cx} y1={cy} x2={cx + r * Math.cos(a)} y2={cy + r * Math.sin(a)} stroke="rgba(255,255,255,0.06)" strokeWidth={1} />; })}
   224→      <path d={pathData} fill="rgba(62,180,137,0.15)" stroke="#3EB489" strokeWidth={1.5} />
   225→      {points.map((p, i) => <circle key={i} cx={p.x} cy={p.y} r={3} fill="#3EB489" />)}
   226→      {entries.map(([name], i) => { const a = (Math.PI * 2 * i) / entries.length - Math.PI / 2; return <text key={i} x={cx + (r + 12) * Math.cos(a)} y={cy + (r + 12) * Math.sin(a)} textAnchor="middle" dominantBaseline="middle" className="fill-zinc-400 text-[7px]" style={{ fontFamily: 'var(--font-vazirmatn)' }}>{name}</text>; })}
   227→    </svg>
   228→  );
   229→}

   335→        </DialogHeader>
   336→
   337→        <div className="space-y-4">
   338→          {/* Subject */}
   339→          <div>
   340→            <label className="text-xs text-zinc-400 mb-1 block">درس</label>
   341→            <select
   342→              value={subject}
   343→              onChange={(e) => handleSubjectChange(e.target.value)}

   355→            <label className="text-xs text-zinc-400 mb-1 block">مبحث</label>

   389→   389→  const [newSubject, setNewSubject] = useState('');

   393→   393→
   394→   394→  // Filter cards based on all criteria
   395→   395→  const filteredCards = flashcards.filter((c) => {
   396→   396→    if (tab === 'marked') {
   397→   397→      // Show only marked cards in the "marked" tab
   398→   398→      if (c.mastery === 'mastered' && masteryFilter === 'all') return true;
   399→   399→      if (c.mastery === 'review' && masteryFilter === 'all') return true;
   400→   400→      if (c.mastery === 'weak' && masteryFilter === 'all') return true;
   401→   401→      if (masteryFilter !== 'all' && c.mastery !== masteryFilter) return false;
   402→   402→    }
   403→   403→    // Subject filter
   404→   404→    if (selectedSubject && c.subject !== selectedSubject) return false;
   405→   405→    // Mastery filter
   406→   406→    if (masteryFilter !== 'all' && c.mastery !== masteryFilter) return false;
   407→   407→    return true;
   408→   408→  });
   409→   409→
   410→   410→  const currentCard = filteredCards[currentIndex] || null;
   411→   411→
   412→   412→  const handleFlip = useCallback(() => {
   413→   413→    setIsFlipped((prev) => !prev);
   414→   414→  }, []);
   415→   415→
   416→   416→  const handleMastery = useCallback(
   417→   417→    (mastery: Flashcard['mastery']) => {
   418→   418→      if (!currentCard) return;
   419→   419→      updateFlashcard(currentCard.id, { mastery });
   420→   420→      setIsFlipped(false);
   421→   421→      setTimeout(() => {
   422→   422→        if (currentIndex < filteredCards.length - 1) {
   423→   423→          setCurrentIndex((prev) => prev + 1);
   424→   424→        }
   425→   425→      }, 300);
   426→   426→    },
   427→   427→    [currentCard, updateFlashcard, currentIndex, filteredCards.length]
   428→   428→  );
   429→   429→
   430→   430→  const handlePrev = useCallback(() => {
   431→   431→    setIsFlipped(false);
   432→   432→    setCurrentIndex((prev) => Math.max(0, prev - 1));
   433→   433→  }, []);
   434→   434→
   435→   435→  const handleNext = useCallback(() => {
   436→   436→    setIsFlipped(false);
   437→   437→    setCurrentIndex((prev) => Math.min(filteredCards.length - 1, prev + 1));
   438→   438→  }, [filteredCards.length]);
   439→   439→
   440→   440→  const handleAddCard = useCallback(() => {
   441→   441→    if (!newFront.trim() || !newBack.trim()) {
   442→   442→      toast('لطفاً هر دو طرف کارت رو پر کن');

   805→   805→function PomodoroTool() {
   806→   806→  const {
   807→   807→    pomodoroTime,
   808→   808→    pomodoroRunning,
   809→   809→    pomodoroMode,
   810→   810→    setPomodoroTime,

   815→   815→
   816→   816→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   817→   817→
   818→   818→  useEffect(() => {
   819→   819→    if (pomodoroRunning) {
   820→   820→      intervalRef.current = setInterval(() => {
   821→   821→        const currentTime = useAppStore.getState().pomodoroTime;
   822→   822→        if (currentTime <= 1) {
   823→   823→          setPomodoroRunning(false);

   825→   825→            toast('آفرین! وقت استراحت 🎉');

   830→   830→            setPomodoroMode('work');
   831→   831→            setPomodoroTime(25 * 60);
   832→   832→          }
   833→   833→        } else {
   834→   834→          setPomodoroTime(currentTime - 1);
   835→   835→        }
   836→   836→      }, 1000);
   837→   837→    }
   838→   838→
   839→   839→    return () => {
   840→   840→      if (intervalRef.current) {
   841→   841→        clearInterval(intervalRef.current);
   842→   842→        intervalRef.current = null;
   843→   843→      }
   844→   844→    };
   845→   845→  }, [pomodoroRunning, pomodoroMode, setPomodoroTime, setPomodoroRunning, setPomodoroMode]);
   846→   846→

    26→    26→function toPersianDigits(num: number): string {
    27→    27→  const persianDigits = ['۰', '۱', '۲', '۳', '۴', '۵', '۶', '۷', '۸', '۹'];
    28→    28→  return num
    29→    29→    .toString()
    30→    30→    .split('')
    31→    31→    .map((d) => persianDigits[parseInt(d)] ?? d)
    32→    32→    .join('');
    33→    33→}

   171→   171→// ===== Counter Animation Hook =====
   172→   172→function useCounter(end: number, duration = 2000, startOnView = true) {
   173→   173→  const [count, setCount] = useState(0);
   174→   174→  const ref = useRef(null);
   175→   175→  const isInView = useInView(ref, { once: true });
   176→   176→
   177→   177→  useEffect(() => {
   178→   178→    if (!startOnView || !isInView) return;
   179→   179→
   180→   180→    let startTime: number | null = null;
   181→   181→    const step = (timestamp: number) => {
   182→   182→      if (!startTime) startTime = timestamp;
   183→   183→      const progress = Math.min((timestamp - startTime) / duration, 1);
   184→   184→      const eased = 1 - Math.pow(1 - progress, 3); // ease-out cubic
   185→   185→      setCount(Math.floor(eased * end));
   186→   186→      if (progress < 1) {
   187→   187→        requestAnimationFrame(step);
   188→   188→      }
   189→   189→    };
   190→   190→    requestAnimationFrame(step);
   191→   191→  }, [end, duration, isInView, startOnView]);
   192→   192→
   193→   193→  return { count, ref };
   194→   194→}
